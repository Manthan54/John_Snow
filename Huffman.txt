
import java.util.*;

class Node {
    char ch;
    int freq;
    Node left, right;

    Node(char ch, int freq) {
        this.ch = ch;
        this.freq = freq;
        left = right = null;
    }
}

public class Main {

    static void generateCodes(Node root, String code,
                             Map<Character, String> huffmanCode) {
        if (root == null)
            return;

        // Leaf node represents a character
        if (root.left == null && root.right == null) {
            huffmanCode.put(root.ch, code);
            return;
        }

        generateCodes(root.left, code + "0", huffmanCode);
        generateCodes(root.right, code + "1", huffmanCode);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a string: ");
        String str = sc.nextLine();

        if (str.isEmpty()) {
            System.out.println("Empty string cannot be encoded.");
            return;
        }

        // Step 1: Calculate character frequencies
        Map<Character, Integer> freq = new TreeMap<>();

        for (char ch : str.toCharArray()) {
            freq.put(ch, freq.getOrDefault(ch, 0) + 1);
        }

        // Step 2: Create Min Heap
        PriorityQueue<Node> pq = new PriorityQueue<>(
            (a, b) -> Integer.compare(a.freq, b.freq)
        );

        for (Map.Entry<Character, Integer> p : freq.entrySet()) {
            pq.add(new Node(p.getKey(), p.getValue()));
        }

        // Step 3: Build Huffman Tree
        while (pq.size() > 1) {
            Node left = pq.poll();
            Node right = pq.poll();

            Node parent = new Node('\0',
                                   left.freq + right.freq);

            parent.left = left;
            parent.right = right;

            pq.add(parent);
        }

        Node root = pq.poll();

        // Step 4: Generate Huffman Codes
        Map<Character, String> huffmanCode = new TreeMap<>();
        generateCodes(root, "", huffmanCode);

        // Step 5: Display frequencies and codes
        System.out.println("\nCharacter Frequencies and Huffman Codes:");

        for (Map.Entry<Character, Integer> p : freq.entrySet()) {
            System.out.println(p.getKey() + " : " +
                p.getValue() + " : " +
                huffmanCode.get(p.getKey()));
        }

        // Step 6: Encode the original string
        String encoded = "";

        for (char ch : str.toCharArray()) {
            encoded += huffmanCode.get(ch);
        }

        System.out.println("\nOriginal String: " + str);
        System.out.println("Encoded String: " + encoded);
        System.out.println("Original Size: " +
                           str.length() * 8 + " bits");
        System.out.println("Compressed Size: " +
                           encoded.length() + " bits");

        sc.close();
    }
}
